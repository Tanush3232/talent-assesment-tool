  // Admin Form state values (Impact Level initialized to 'Medium' but open-ended text input)
  const [adminForm, setAdminForm] = useState({
    id: '', name: '', email: '', role: '', impactLevel: 'Medium',
    entity: 'Zuari Industries Ltd', department: 'Engineering & R&D',
    location: 'Bengaluru Outer Ring Road CoE', managerId: 'MGR001',
    managerEmail: 'sandeep.sharma@zuari.com'
  });

  // 2. Pure Helper Functions
  const calculateSectionScore = (section, scoresObj = tempScores) => {
    const indicators = section === 'ability' ? ['1.1', '1.2', '1.3', '1.4'] :
                       section === 'aspiration' ? ['2.1', '2.2', '2.3', '2.4'] :
                       ['3.1', '3.2', '3.3', '3.4'];
    
    let total = 0;
    let counted = 0;
    indicators.forEach(id => {
      const val = scoresObj[section] && scoresObj[section][id];
      if (val !== undefined && val !== null && val !== '') {
        total += Number(val);
        counted++;
      }
    });
    return { sum: total, counted, totalInSchema: indicators.length };
  };

  const getTalentClassification = (totalScore, lowScoresCount) => {
    if (totalScore === 0) return { category: 'Pending Rating', description: 'Complete evaluation indicators to compute classification.' };
    
    if (lowScoresCount > 2) {
      if (totalScore >= 37) {
        return {
          category: 'Promotable/Expandable',
          description: 'Excluded from High Potential (HiPo) status due to scoring below 3 on more than 2 sub-dimensions. Strong performance aggregate, but requires targeted development on specific behavior blocks.',
          forcedExclusion: true
        };
      }
    }

    if (totalScore >= 37) {
      return {
        category: 'High Potential (HiPo)',
        description: 'Has the potential to move 1-2 levels up. Excels across all three pillars. Can be considered for promotion immediately, if other promotion criteria are met. Should receive broad organizational exposure, executive mentorship, and rapid development paths.'
      };
    } else if (totalScore >= 24) {
      return {
        category: 'Promotable/Expandable',
        description: 'Has the potential to move 1 level up with targeted development. Can be considered for larger roles at the same responsibility level immediately. Should receive coaching/mentoring to address the specific gaps on aspects of ability, aspiration or leadership.'
      };
    } else {
      return {
        category: 'Well-placed',
        description: 'Not yet ready for larger roles. Consider job enrichment, deep specialization, and retention if the performance is good.'
      };
    }
  };

  const isFormReadOnly = selectedEmployee ? (selectedEmployee.status === 'Completed' || currentUser?.type === 'hr') : false;

  const triggerToast = (message, type = 'success') => {
    setToast({ message, type });
    setTimeout(() => setToast(null), 4000);
  };

  // 3. Memoized Aggregations
  const availableManagers = useMemo(() => {
    const ids = [...new Set(employees.map(emp => emp.managerId))];
    return ids.map(id => USERS[id] || { id, name: id }).filter(Boolean);
  }, [employees]);

  const functionSummaries = useMemo(() => {
    const summaries = {
      'Technology': { name: 'Technology', total: 0, completed: 0, hipos: 0 },
      'Product': { name: 'Product', total: 0, completed: 0, hipos: 0 },
      'Commercial': { name: 'Commercial', total: 0, completed: 0, hipos: 0 },
      'Operations': { name: 'Operations', total: 0, completed: 0, hipos: 0 }
    };

    employees.forEach(emp => {
      const func = DEPARTMENT_TO_FUNCTION[emp.department] || 'Operations';
      summaries[func].total += 1;
      if (emp.status === 'Completed') {
        summaries[func].completed += 1;
        const ab = calculateSectionScore('ability', emp.scores);
        const asp = calculateSectionScore('aspiration', emp.scores);
        const lead = calculateSectionScore('leadership', emp.scores);
        const grandTotal = ab.sum + asp.sum + lead.sum;

        let lowCount = 0;
        const allScoresFlat = { ...emp.scores.ability, ...emp.scores.aspiration, ...emp.scores.leadership };
        Object.values(allScoresFlat).forEach(val => {
          if (val !== undefined && val !== null && Number(val) < 3) {
            lowCount++;
          }
        });
        const classification = getTalentClassification(grandTotal, lowCount);
        if (classification.category === 'High Potential (HiPo)' && !classification.forcedExclusion) {
          summaries[func].hipos += 1;
        }
      }
    });
    return Object.values(summaries);
  }, [employees]);

  const departmentSummaries = useMemo(() => {
    const summaries = {};
    const depts = [...new Set(employees.map(emp => emp.department))];
    depts.forEach(d => {

                  <label className="text-[10px] text-slate-400 uppercase font-mono tracking-widest font-bold block mb-1">Employee Entity</label>
                  <input 
                    type="text"
                    value={adminForm.entity}
                    onChange={(e) => setAdminForm({...adminForm, entity: e.target.value})}
                    placeholder="Zuari Industries Ltd"
                    className="w-full p-2.5 bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-slate-900 focus:bg-white text-xs"
                  />
                </div>
              </div>

              <div className="grid grid-cols-3 gap-4">
                <div>
                  <label className="text-[10px] text-slate-400 uppercase font-mono tracking-widest font-bold block mb-1">Manager ID</label>
                  <select
                    value={adminForm.managerId}
                    onChange={(e) => setAdminForm({...adminForm, managerId: e.target.value})}
                    className="w-full p-2.5 bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-slate-955 focus:bg-white text-xs cursor-pointer"
                  >
                    {Object.values(USERS).filter(u => u.type === 'manager').map(mgr => (
                      <option key={mgr.id} value={mgr.id}>{mgr.name} ({mgr.id})</option>
                    ))}
                  </select>
                </div>
                <div className="col-span-2">
                  <label className="text-[10px] text-slate-400 uppercase font-mono tracking-widest font-bold block mb-1">Manager Email</label>
                  <input 
                    type="email"
                    value={adminForm.managerEmail}
                    onChange={(e) => setAdminForm({...adminForm, managerEmail: e.target.value})}
                    placeholder="sandeep.sharma@zuari.com"
                    className="w-full p-2.5 bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-slate-900 focus:bg-white text-xs"
                  />
                </div>
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="text-[10px] text-slate-400 uppercase font-mono tracking-widest font-bold block mb-1">Department</label>
                  <select
                    value={adminForm.department}
                    onChange={(e) => setAdminForm({...adminForm, department: e.target.value})}
                    className="w-full p-2.5 bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-slate-955 focus:bg-white text-xs cursor-pointer"
                  >
                    <option value="Engineering & R&D">Engineering & R&D</option>
                    <option value="Product Management">Product Management</option>
                    <option value="Design Studio">Design Studio</option>
                    <option value="Sales & Commercial">Sales & Commercial</option>
                    <option value="Business Enablement">Business Enablement</option>
                  </select>
                </div>
                <div>
                  <label className="text-[10px] text-slate-400 uppercase font-mono tracking-widest font-bold block mb-1">Location</label>
                  <input 
                    type="text"
                    value={adminForm.location}
                    onChange={(e) => setAdminForm({...adminForm, location: e.target.value})}
                    placeholder="Bengaluru Outer Ring Road CoE"
                    className="w-full p-2.5 bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-slate-900 focus:bg-white text-xs"
                  />
                </div>
              </div>

              <div className="pt-4 border-t border-slate-100 flex justify-end gap-2 bg-slate-50 -mx-6 -mb-6 p-4 rounded-b-2xl">
                <button 
                  type="button" 
                  onClick={() => setIsAdminCreateOpen(false)}
                  className="px-4 py-2 hover:bg-slate-100 text-slate-700 rounded-lg text-xs font-bold transition border border-slate-200 bg-white"
                >
                  Cancel
                </button>
                <button 
                  type="submit"
                  className="px-4 py-2 bg-slate-950 hover:bg-slate-900 text-white rounded-lg text-xs font-bold transition shadow-md"
                >
                  {adminSelectedEmp ? "Save Changes" : "Register Employee"}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ADMIN BULK UPLOAD SLIDE OVER MODAL */}
      {isAdminBulkOpen && currentUser.type === 'admin' && (
        <div className="fixed inset-0 z-50 bg-slate-900/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl shadow-2xl border border-slate-100 max-w-2xl w-full max-h-[85vh] overflow-y-auto flex flex-col animate-in fade-in zoom-in-95 duration-150">
            <div className="p-5 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
              <div className="flex items-center gap-2">
                <Upload className="w-5 h-5 text-rose-500" />
                <h3 className="font-serif font-bold text-slate-900 text-base">Bulk Upload CSV / Raw Text Data</h3>
              </div>
              <button onClick={() => setIsAdminBulkOpen(false)} className="text-slate-400 hover:text-slate-600 bg-slate-200/50 hover:bg-slate-200 p-1.5 rounded-lg transition">✕</button>
            </div>

            <div className="p-6 space-y-4 text-xs">
              <div className="bg-slate-50 p-4 rounded-xl border border-slate-200 space-y-2 text-slate-600">
                <span className="font-bold text-slate-900 block text-[10px] uppercase font-mono tracking-wide">Pasting Instructions:</span>
                <p>Paste comma-separated (CSV) or tab-separated (TSV) rows directly. If the Employee ID matches an existing record, the system will update it. Otherwise, a new record is added.</p>
                <span className="font-bold text-slate-900 block text-[10px] uppercase font-mono tracking-wide mt-3">Expected column sequence (No Quotes Required):</span>
                <code className="block bg-slate-200 p-2 rounded text-[10px] text-slate-800 break-all select-all">
                  Emp ID, Employee Name, Employee Email, Designation, Impact Level, Employee Entity, Manager ID, Manager Email, Location, Department
                </code>
              </div>

              <textarea 
                rows={10}
                value={adminBulkText}
                onChange={(e) => setAdminBulkText(e.target.value)}
                placeholder="EMP015, John Doe, john.doe@zuari.com, Senior Developer, High, Zuari Industries Ltd, MGR001, sandeep.sharma@zuari.com, Bengaluru ORR CoE, Engineering & R&D"
                className="w-full p-3 font-mono text-[11px] bg-slate-50 hover:bg-slate-100/50 focus:bg-white border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-slate-900 transition"
              />

              <div className="flex items-center justify-between">
                <button
                  type="button"
                  onClick={() => {
                    setAdminBulkText("EMP015, Jane Miller, jane.miller@zuari.com, Senior Project Lead, Critical, Zuari Industries Ltd, MGR002, divya.nair@zuari.com, Mumbai BKC Head Office, Business Enablement\nEMP016, Amit Singh, amit.singh@zuari.com, Software Specialist, Medium, Adventz Corporate CoE, MGR001, sandeep.sharma@zuari.com, Hyderabad HITEC City, Engineering & R&D");
                  }}
                  className="text-xs text-rose-600 hover:text-rose-800 hover:underline font-semibold"
                >
                  Load Sample Paste Rows
                </button>
                <span className="text-[10px] text-slate-400">Total lines: {adminBulkText.split('\n').filter(Boolean).length}</span>
              </div>
            </div>

            <div className="p-4 border-t border-slate-100 bg-slate-50 flex justify-end gap-2 rounded-b-2xl">
              <button 
                type="button" 
                onClick={() => setIsAdminBulkOpen(false)}
                className="px-4 py-1.5 hover:bg-slate-100 text-slate-700 border border-slate-200 bg-white rounded-lg text-xs font-bold transition"
              >
                Cancel
              </button>
              <button 
                type="button"
                onClick={handleBulkCSVUpload}
                className="px-4 py-1.5 bg-rose-600 hover:bg-rose-700 text-white rounded-lg text-xs font-bold transition shadow-md"
              >
                Parse & Upload Data
              </button>
            </div>
          </div>
        </div>
      )}

      {isGuidelinesModalOpen && (
        <div className="fixed inset-0 z-50 bg-slate-900/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl shadow-2xl border border-slate-100 max-w-2xl w-full max-h-[85vh] overflow-y-auto flex flex-col animate-in fade-in zoom-in-95 duration-150">
            <div className="p-6 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
              <div className="flex items-center gap-2.5">
                <BookOpen className="w-5 h-5 text-slate-800" />
                <div>
                  <h3 className="font-serif font-bold text-slate-900 text-base">Potential Assessment Guidelines</h3>
                  <p className="text-[10px] text-slate-400 uppercase font-mono tracking-widest text-left">Calibration Framework V2.0</p>
                </div>
              </div>
              <button onClick={() => setIsGuidelinesModalOpen(false)} className="text-slate-400 hover:text-slate-600 bg-slate-200/50 hover:bg-slate-200 p-1.5 rounded-lg transition">✕</button>
            </div>
            
            <div className="p-6 space-y-5 overflow-y-auto text-xs leading-relaxed text-slate-600 text-left">
              <div>
                <h4 className="font-bold text-slate-900 mb-1.5">1. Purpose & Scope of Potential Calibration</h4>
                <p>The purpose of this tool is to enable people managers to assess the potential of their team members in a structured and consistent manner. Potential can be estimated by assessing the employee on the various behavioural indicators of ability, aspiration and leadership that are given below.</p>
              </div>

              <div>
                <h4 className="font-bold text-slate-900 mb-1.5">2. Behavioral Evaluation Scale (1–4)</h4>
                <div className="grid grid-cols-2 gap-3 mt-2">
                  <div className="p-3 bg-slate-50 rounded-lg border border-slate-100">
                    <span className="font-bold text-slate-800 block text-[11px] mb-0.5">Rating 1: Rarely or never</span>
                    <span className="text-slate-500 block text-[10px]">Rarely or never displays the behaviour.</span>
                  </div>
                  <div className="p-3 bg-slate-50 rounded-lg border border-slate-100">
                    <span className="font-bold text-slate-800 block text-[11px] mb-0.5">Rating 2: Sometimes</span>
                    <span className="text-slate-500 block text-[10px]">Demonstrates the behaviour sometimes.</span>
                  </div>
                  <div className="p-3 bg-slate-50 rounded-lg border border-slate-100">
                    <span className="font-bold text-slate-800 block text-[11px] mb-0.5">Rating 3: Most of the time</span>
                    <span className="text-slate-500 block text-[10px]">Demonstrates the behaviour most of the time.</span>
                  </div>
                  <div className="p-3 bg-slate-50 rounded-lg border border-slate-100">
                    <span className="font-bold text-slate-800 block text-[11px] mb-0.5">Rating 4: Consistently</span>
                    <span className="text-slate-500 block text-[10px]">Demonstrates the behaviour consistently/role models.</span>
                  </div>
                </div>
              </div>

              <div className="p-4 bg-indigo-50 border border-indigo-100 rounded-xl">
                <h4 className="font-bold text-indigo-900 flex items-center gap-1.5 mb-1.5 text-xs">
                  <AlertTriangle className="w-4 h-4 text-indigo-700" />
                  Talent Progression Exclusion Rules
                </h4>
                <p className="text-indigo-955 text-[11px] leading-relaxed">
                  Any employee who scores below 3 on more than two indicators (across Ability, Aspiration, and Leadership) is automatically excluded from High Potential (HiPo) classification.
                </p>
              </div>
            </div>

            <div className="p-4 border-t border-slate-100 bg-slate-50 flex justify-end">
              <button onClick={() => setIsGuidelinesModalOpen(false)} className="px-4 py-1.5 bg-slate-900 hover:bg-slate-800 text-white rounded-lg text-xs font-bold transition">Close</button>
            </div>
          </div>
        </div>
      )}

      {isLowScoreModalOpen && (
        <div className="fixed inset-0 z-50 bg-slate-900/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl shadow-2xl border border-slate-100 max-w-4xl w-full max-h-[85vh] overflow-y-auto flex flex-col animate-in fade-in zoom-in-95 duration-150">
            <div className="p-6 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
              <div className="text-left">
                <h3 className="font-serif font-bold text-slate-900 text-base">Indicators Scoring Below Rating 3</h3>
                <p className="text-[10px] text-slate-400 uppercase font-mono tracking-widest mt-0.5">Detailed inspection sheet for behavior evidence</p>
              </div>
              <button onClick={() => setIsLowScoreModalOpen(false)} className="text-slate-400 hover:text-slate-600 bg-slate-200/50 hover:bg-slate-200 p-1.5 rounded-lg transition">✕</button>
            </div>
            
            <div className="p-6 overflow-y-auto text-left">
              {lowScoreRecordsList.length > 0 ? (
                <div className="overflow-x-auto rounded-xl border border-slate-200 bg-white">
                  <table className="w-full text-left border-collapse text-xs">
                    <thead>
                      <tr className="bg-slate-50 border-b border-slate-200 text-slate-700 font-bold">
                        <th className="p-3 w-1/6">Section</th>
                        <th className="p-3 w-1/4">Subdimension</th>
                        <th className="p-3 w-1/12 text-center">Rating</th>
                        <th className="p-3">Evidence / Justification Text</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-100">
                      {lowScoreRecordsList.map((item) => (
                        <tr key={item.id} className={`${item.rating < 2 ? 'bg-rose-50/30' : 'hover:bg-slate-50/50'}`}>
                          <td className="p-3"><span className="font-semibold text-slate-700">{item.section}</span></td>
                          <td className="p-3">
                            <div className="space-y-1">
                              <span className="inline-block px-1.5 py-0.5 bg-slate-100 rounded text-[9px] font-mono font-bold text-slate-600">Indicator {item.id}</span>
                              <span className="block font-medium text-slate-900">{item.title}</span>
                            </div>
                          </td>
                          <td className="p-3 text-center">
                            <span className="inline-block w-7 h-7 leading-7 rounded-lg font-mono font-bold text-xs bg-amber-50 text-amber-700 border border-amber-200">{item.rating}</span>
                          </td>
                          <td className="p-3 text-slate-600 leading-relaxed text-[11px] italic">"{item.evidence}"</td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              ) : (
                <div className="text-center py-12 text-slate-400">
                  <span className="font-serif text-slate-400 text-base block mb-1">No indicators rated under 3</span>
                  <p className="text-xs text-slate-500">All behavioral dimensions for this employee are rated 3 or 4.</p>
                </div>
              )}
            </div>

            <div className="p-4 border-t border-slate-100 bg-slate-50 flex justify-end">
              <button onClick={() => setIsLowScoreModalOpen(false)} className="px-4 py-1.5 bg-[#0f172a] hover:bg-slate-800 text-white rounded-lg text-xs font-bold transition">Close</button>
            </div>
          </div>
        </div>
      )}

    </div>
  );
}
